## 🔷 Laços e Input/Output de Dados

Vamos explorar como usar loops para **ler dados do usuário**, **processar em tempo real** e **gerar saídas formatadas**. Essas técnicas são essenciais para criar sistemas interativos e relatórios dinâmicos.

## 🔷 1) **Input básico dentro de loops**

### ✅ **Coletar dados de múltiplos candidatos**:

```python
candidatos = []
quantidade = int(input("Quantos candidatos deseja cadastrar? "))

for i in range(quantidade):
    print(f"\nCandidato {i+1}:")
    nome = input("Nome: ")
    idade = int(input("Idade: "))
    experiencia = int(input("Anos de experiência: "))
    
    candidatos.append({
        "nome": nome,
        "idade": idade,
        "experiencia": experiencia
    })

print(f"\n✅ {len(candidatos)} candidatos cadastrados!")
```

### ✅ **Validação de entrada com `while`**:

```python
notas = []
print("Digite as notas (0-10) ou 'sair' para terminar:")

while True:
    entrada = input("Nota: ")
    
    if entrada.lower() == "sair":
        break
    
    try:
        nota = float(entrada)
        if 0 <= nota <= 10:
            notas.append(nota)
            print(f"Nota {nota} adicionada!")
        else:
            print("❌ Nota deve estar entre 0 e 10")
    except ValueError:
        print("❌ Digite um número válido")

print(f"\nNotas coletadas: {notas}")
```

## 🔷 2) **Processamento em tempo real com feedback**

### ✅ **Sistema de triagem RH com progresso**:

```python
candidatos = [
    {"nome": "Ana", "idade": 25, "experiencia": 3},
    {"nome": "Bruno", "idade": 17, "experiencia": 1},

    else:
        reprovados.append(candidato["nome"])
        print(f"  Resultado: ❌ REPROVADO")

print("\n" + "=" * 40)
print(f"📊 RESUMO: {len(aprovados)} aprovados, {len(reprovados)} reprovados")
```

## 🔷 4) **Menu interativo com loop**

### ✅ **Sistema de gestão QA**:

```python
def menu_qa():
    bugs = []
    
    while True:
        print("\n" + "=" * 40)
        print("🐛 SISTEMA DE GESTÃO QA")
        print("=" * 40)
        print("1. Reportar novo bug")
        print("2. Listar bugs")
        print("3. Estatísticas")
        print("4. Sair")
        
        opcao = input("\nEscolha uma opção (1-4): ")
        
        if opcao == "1":
            print("\n📝 REPORTAR BUG")
            descricao = input("Descrição: ")
            severidade = input("Severidade (CRITICO/ALTO/MEDIO/BAIXO): ").upper()
            
            bugs.append({
                "id": len(bugs) + 1,
                "descricao": descricao,
                "severidade": severidade,
                "status": "ABERTO"
            })
            print("✅ Bug reportado com sucesso!")
            
        elif opcao == "2":
            print("\n📋 BUGS REPORTADOS")
            if not bugs:
                print("Nenhum bug reportado.")
            else:
                for bug in bugs:
                    print(f"  #{bug['id']} - {bug['descricao']} [{bug['severidade']}]")
                    
        elif opcao == "3":
            print("\n📊 ESTATÍSTICAS")
            if bugs:
                total = len(bugs)
                criticos = sum(1 for b in bugs if b["severidade"] == "CRITICO")
                print(f"Total bugs: {total}")
                print(f"Bugs críticos: {criticos} ({criticos/total*100:.1f}%)")
            else:
                print("Nenhum dado disponível.")
                
        elif opcao == "4":
            print("\n👋 Encerrando sistema...")
            break
            
        else:
            print("❌ Opção inválida!")

# Executar menu
menu_qa()
```

## 🔷 5) **Leitura de arquivos com loops**

### ✅ **Processar arquivo CSV de bugs**:

```python
# Simulando leitura de arquivo
dados_bugs = """ID,SEVERIDADE,STATUS,DESCRICAO
1,CRITICO,ABERTO,Login falha
2,ALTO,FECHADO,Pagamento lento
3,MEDIO,ABERTO,UI quebrada
4,CRITICO,ABERTO,Perda de dados"""

print("📄 Processando arquivo de bugs...")
print("-" * 50)

# Processar linha por linha
linhas = dados_bugs.strip().split('\n')
bugs_processados = []

for i, linha in enumerate(linhas):
    if i == 0:  # Cabeçalho
        print(f"Cabeçalho: {linha}")
        continue
    
    # Processar dados
    campos = linha.split(',')
    bug = {
        "id": int(campos[0]),
        "severidade": campos[1],
        "status": campos[2],
        "descricao": campos[3]
    }
    bugs_processados.append(bug)
    
    print(f"Processado bug #{bug['id']}: {bug['descricao']}")

print(f"\n✅ {len(bugs_processados)} bugs processados do arquivo")
```

## 🔷 6) **Geração de relatórios com loops**

### ✅ **Relatório HTML simples**:

```python
def gerar_relatorio_html(bugs):
    """Gera relatório HTML a partir de lista de bugs"""
    
    html = []
    html.append("<html><head><title>Relatório QA</title></head><body>")
    html.append("<h1>🐛 Relatório de Bugs</h1>")
    html.append("<table border='1'>")
    html.append("<tr><th>ID</th><th>Severidade</th><th>Status</th><th>Descrição</th></tr>")
    
    for bug in bugs:
        # Cor baseada na severidade
        cor = "red" if bug["severidade"] == "CRITICO" else \
              "orange" if bug["severidade"] == "ALTO" else \
              "yellow" if bug["severidade"] == "MEDIO" else "green"
        
        html.append(f"<tr style='color: {cor}'>")
        html.append(f"<td>{bug['id']}</td>")
        html.append(f"<td>{bug['severidade']}</td>")
        html.append(f"<td>{bug['status']}</td>")
        html.append(f"<td>{bug['descricao']}</td>")
        html.append("</tr>")
    
    html.append("</table>")
    
    # Estatísticas
    criticos = sum(1 for b in bugs if b["severidade"] == "CRITICO")
    html.append(f"<p><strong>Total bugs:</strong> {len(bugs)}</p>")
    html.append(f"<p><strong>Bugs críticos:</strong> {criticos}</p>")
    
    html.append("</body></html>")
    
    return "\n".join(html)

# Exemplo de uso
bugs = [
    {"id": 1, "severidade": "CRITICO", "status": "ABERTO", "descricao": "Login falha"},
    {"id": 2, "severidade": "ALTO", "status": "FECHADO", "descricao": "Pagamento lento"},
]

relatorio = gerar_relatorio_html(bugs)
print("📋 Relatório HTML gerado:")
print(relatorio[:200] + "...")  # Mostra apenas o início
```

## 🔷 7) **Input com validação avançada**

### ✅ **Sistema de avaliação de desempenho**:

```python
def coletar_avaliacoes():
    """Coleta avaliações de desempenho com validação"""
    
    avaliacoes = []
    funcionarios = ["Ana", "Bruno", "Carla", "Daniel"]
    
    print("📊 SISTEMA DE AVALIAÇÃO DE DESEMPENHO")
    print("Avalie cada funcionário de 1 a 5")
    print("-" * 40)
    
    for funcionario in funcionarios:
        while True:
            try:
                entrada = input(f"\n{funcionario} (1-5): ")
                
                if entrada.lower() == "pular":
                    print(f"⏭️  {funcionario} pulado")
                    break
                
                nota = float(entrada)
                
                if 1 <= nota <= 5:
                    avaliacoes.append({
                        "funcionario": funcionario,
                        "nota": nota
                    })
                    print(f"✅ Nota {nota} registrada para {funcionario}")
                    break
                else:
                    print("❌ Nota deve estar entre 1 e 5")
                    
            except ValueError:
                print("❌ Digite um número válido ou 'pular'")
    
    return avaliacoes

# Coletar dados
dados = coletar_avaliacoes()

# Processar resultados
if dados:
    print("\n" + "=" * 40)
    print("📈 RESULTADOS DA AVALIAÇÃO")
    
    total_notas = sum(item["nota"] for item in dados)
    media = total_notas / len(dados)
    
    print(f"\nMédia geral: {media:.2f}")
    print("\nNotas individuais:")
    for item in dados:
        print(f"  {item['funcionario']}: {item['nota']}")
else:
    print("\n⚠️  Nenhuma avaliação registrada")
```

## 🔷 8) **Output progressivo com `sys.stdout`**

### ✅ **Barra de progresso durante processamento**:

```python
import time
import sys

def processar_testes_lento(total_testes=10):
    """Simula processamento lento com barra de progresso"""
    
    print("🚀 Executando testes...")
    print("[", end="")
    
    for i in range(total_testes):
        # Simula trabalho
        time.sleep(0.5)
        
        # Atualiza barra de progresso
        progresso = (i + 1) / total_testes * 50  # 50 caracteres de largura
        sys.stdout.write("█" * int(progresso // (50 / total_testes)))
        sys.stdout.flush()
    
    print("] 100%")
    print(f"✅ {total_testes} testes executados!")

# Executar
processar_testes_lento(5)
```
