## 🔷 **5. Análise 1: Candidatos por Estado (MapReduce)**

Vamos contar quantos candidatos temos por estado.

### ✅ **Código Java (CandidatosPorEstado.java)**:

```java
import java.io.IOException;
import java.util.StringTokenizer;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.IntWritable;
import org.apache.hadoop.io.Text;
import org.apache.hadoop.mapreduce.Job;
import org.apache.hadoop.mapreduce.Mapper;
import org.apache.hadoop.mapreduce.Reducer;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;

public class CandidatosPorEstado {
    
    // MAP: Extrai estado de cada linha
    public static class TokenizerMapper 
        extends Mapper<Object, Text, Text, IntWritable>{
        
        private final static IntWritable one = new IntWritable(1);
        private Text estado = new Text();
        
        public void map(Object key, Text value, Context context
                       ) throws IOException, InterruptedException {
            
            // Ignorar cabeçalho
            if (value.toString().contains("id,nome")) {
                return;
            }
            
            String[] campos = value.toString().split(",");
            if (campos.length >= 4) {
                estado.set(campos[3]); // Coluna estado
                context.write(estado, one);
            }
        }
    }
    
    // REDUCE: Soma contagens por estado
    public static class IntSumReducer 
        extends Reducer<Text, IntWritable, Text, IntWritable> {
        
        private IntWritable resultado = new IntWritable();
        
        public void reduce(Text key, Iterable<IntWritable> values,
                          Context context
                          ) throws IOException, InterruptedException {
            
            int soma = 0;
            for (IntWritable val : values) {
                soma += val.get();
            }
            resultado.set(soma);
            context.write(key, resultado);
        }
    }
    
    // MAIN: Configura e executa job
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        Job job = Job.getInstance(conf, "candidatos por estado");
        
        job.setJarByClass(CandidatosPorEstado.class);
        job.setMapperClass(TokenizerMapper.class);
        job.setCombinerClass(IntSumReducer.class);
        job.setReducerClass(IntSumReducer.class);
        
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(IntWritable.class);
        
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileOutputFormat.setOutputPath(job, new Path(args[1]));
        
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
```

### ✅ **Compilar e executar**:

```bash

# 2. Criar JAR
```

## 🔷 **6. Análise 2: Bugs por Módulo e Severidade**

Vamos contar bugs por módulo e também por severidade.

### ✅ **Código Python (usando Hadoop Streaming)**

Como Hadoop suporta Python via Streaming API:

**mapper_bugs.py**:

```python
#!/usr/bin/env python3
import sys

# Ler linha por linha do stdin
for linha in sys.stdin:
    # Pular cabeçalho
    if "id_bug,modulo" in linha:
        continue
    
    campos = linha.strip().split(",")
    if len(campos) >= 3:
        modulo = campos[1]
        severidade = campos[2]
        
        # Emitir duas chaves: uma por módulo, outra por severidade
        print(f"{modulo}\t1")
        print(f"{severidade}\t1")
```

**reducer_bugs.py**:

```python
#!/usr/bin/env python3
import sys

chave_atual = None
contagem = 0

for linha in sys.stdin:
    chave, valor = linha.strip().split("\t")
    
    if chave_atual == chave:
        contagem += int(valor)
    else:
        if chave_atual:
            print(f"{chave_atual}\t{contagem}")
        chave_atual = chave
        contagem = int(valor)

# Última chave
if chave_atual:
    print(f"{chave_atual}\t{contagem}")
```

### ✅ **Executar com Hadoop Streaming**:

```bash

  -mapper mapper_bugs.py \
```

## 🔷 **7. Análise 3: Salário Médio por Cargo (MapReduce mais complexo)**

Vamos calcular o salário médio pretendido por cargo.

### ✅ **Código Java (SalarioMedioPorCargo.java)**:

```java
import java.io.IOException;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.DoubleWritable;
import org.apache.hadoop.io.Text;
import org.apache.hadoop.mapreduce.Job;
import org.apache.hadoop.mapreduce.Mapper;
import org.apache.hadoop.mapreduce.Reducer;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;

public class SalarioMedioPorCargo {
    
    // MAP: Emite (cargo, salario)
    public static class SalarioMapper 
        extends Mapper<Object, Text, Text, DoubleWritable> {
        
        private Text cargo = new Text();
        private DoubleWritable salario = new DoubleWritable();
        
        public void map(Object key, Text value, Context context
                       ) throws IOException, InterruptedException {
            
            if (value.toString().contains("id,nome")) {
                return;
            }
            
            String[] campos = value.toString().split(",");
            if (campos.length >= 6) {
                cargo.set(campos[5]); // cargo_pretendido
                try {
                    double sal = Double.parseDouble(campos[6]); // salario_pretendido
                    salario.set(sal);
                    context.write(cargo, salario);
                } catch (NumberFormatException e) {
                    // Ignorar erros de conversão
                }
            }
        }
    }
    
    // REDUCE: Calcula média por cargo
    public static class MediaReducer 
        extends Reducer<Text, DoubleWritable, Text, DoubleWritable> {
        
        private DoubleWritable resultado = new DoubleWritable();
        
        public void reduce(Text key, Iterable<DoubleWritable> values,
                          Context context
                          ) throws IOException, InterruptedException {
            
            double soma = 0;
            int contador = 0;
            
            for (DoubleWritable val : values) {
                soma += val.get();
                contador++;
            }
            
            double media = soma / contador;
            resultado.set(media);
            context.write(key, resultado);
        }
    }
    
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        Job job = Job.getInstance(conf, "salario medio por cargo");
        
        job.setJarByClass(SalarioMedioPorCargo.class);
        job.setMapperClass(SalarioMapper.class);
        job.setReducerClass(MediaReducer.class);
        
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(DoubleWritable.class);
        
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileOutputFormat.setOutputPath(job, new Path(args[1]));
        
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
```

## 🔷 **8. Análise 4: Tempo Médio de Resolução por Desenvolvedor**

Vamos calcular quanto tempo cada desenvolvedor leva em média para resolver bugs.

### ✅ **Código Python (mapper_tempo.py)**:

```python
#!/usr/bin/env python3
import sys

for linha in sys.stdin:
    if "id_bug,modulo" in linha:
        continue
    
    campos = linha.strip().split(",")
    if len(campos) >= 5:
        desenvolvedor = campos[3]
        horas = campos[4]
        
        try:
            horas_float = float(horas)
            print(f"{desenvolvedor}\t{horas_float}")
        except:
            continue
```

### ✅ **Código Python (reducer_tempo.py)**:

```python
#!/usr/bin/env python3
import sys

desenvolvedor_atual = None
soma_horas = 0
contador = 0

for linha in sys.stdin:
    desenvolvedor, horas_str = linha.strip().split("\t")
    horas = float(horas_str)
    
    if desenvolvedor_atual == desenvolvedor:
        soma_horas += horas
        contador += 1
    else:
        if desenvolvedor_atual:
            media = soma_horas / contador
            print(f"{desenvolvedor_atual}\t{media:.2f}")
        desenvolvedor_atual = desenvolvedor
        soma_horas = horas
        contador = 1

if desenvolvedor_atual:
    media = soma_horas / contador
    print(f"{desenvolvedor_atual}\t{media:.2f}")
```
