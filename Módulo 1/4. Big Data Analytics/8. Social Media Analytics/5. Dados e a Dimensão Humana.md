```python
# Código de ética para dados humanos
class DataEthicsFramework:
    def __init__(self):
        self.principles = {
            'transparency': "Clareza sobre coleta e uso de dados",
            'consent': "Consentimento informado e revogável",
            'privacy': "Proteção de dados pessoais",
            'fairness': "Evitar viés e discriminação",
            'accountability': "Responsabilidade pelas decisões baseadas em dados"
        }
    
    def check_bias(self, data, protected_attributes):
        """Verificar viés em dados de RH"""
        biases = {}
        
        for attr in protected_attributes:
            if attr in data.columns:
                # Verificar diferenças significativas
                groups = data[attr].unique()
                if len(groups) > 1:
                    # Exemplo: diferença salarial por gênero
                    salary_by_group = data.groupby(attr)['salario'].mean()
                    max_diff = salary_by_group.max() - salary_by_group.min()
                    
                    if max_diff > salary_by_group.mean() * 0.1:  # >10% diferença
                        biases[attr] = {
                            'difference': max_diff,
                            'ratio': salary_by_group.max() / salary_by_group.min()
                        }
        
        return biases

# Exemplo prático
dados_rh = pd.DataFrame({
    'genero': ['M', 'F', 'M', 'F', 'M', 'F', 'M', 'F'],
    'idade': [30, 28, 35, 32, 40, 38, 45, 42],
    'salario': [7000, 6500, 7500, 6800, 8000, 7200, 8500, 7500],
    'experiencia': [5, 4, 6, 5, 8, 7, 10, 9]
})

ethics = DataEthicsFramework()
biases = ethics.check_bias(dados_rh, ['genero', 'idade'])

print("⚖️ Análise de Viés em Dados de RH:")
for attr, bias_info in biases.items():
    print(f"\n🔍 Viés detectado em '{attr}':")
    print(f"   Diferença máxima: R$ {bias_info['difference']:.2f}")
    print(f"   Razão: {bias_info['ratio']:.2f}:1")
    
    if bias_info['ratio'] > 1.1:
        print("   ⚠️ Diferença significativa detectada!")
```

```python
class HumanDataStoryteller:
    def __init__(self):
        self.story_templates = {
            'turnover': "Entre {date_start} e {date_end}, {n_people} pessoas deixaram a empresa. Isso representa {percentage}% da equipe. A maioria tinha entre {age_range} anos e permaneceu {avg_tenure} meses.",
            'recruitment': "Para preencher {n_positions} vagas, analisamos {n_candidates} candidatos. O processo levou {avg_days} dias em média. {success_rate}% foram contratados.",
            'engagement': "A pesquisa de engajamento mostrou que {happy_percent}% estão satisfeitos, mas {concern_percent}% expressaram preocupações sobre {main_concern}."
        }
    
    def create_turnover_story(self, data):
        """Criar narrativa sobre turnover"""
        story_data = {
            'date_start': data['data_saida'].min().strftime('%d/%m/%Y'),
            'date_end': data['data_saida'].max().strftime('%d/%m/%Y'),
            'n_people': len(data),
            'percentage': (len(data) / 100) * 100,  # Supondo 100 funcionários totais
            'age_range': f"{data['idade'].mean():.0f}-{data['idade'].max()}",
            'avg_tenure': data['tempo_empresa'].mean()
        }
        
        story = self.story_templates['turnover'].format(**story_data)
        
        # Adicionar insights específicos
        if data['motivo_saida'].value_counts().get('salario', 0) > len(data) * 0.3:
            story += "\n\n💡 Insight: Salário foi o principal motivo citado por 30%+ das saídas."
        
        if data['tempo_empresa'].mean() &lt; 12:
            story += "\n💡 Insight: Média de permanência inferior a 1 ano sugere problemas de retenção."
        
        return story
```

```python
                patterns['procrastination'] = "Atraso médio de {:.1f} dias".format(

                )
        
        # Padrão 3: Colaboração
        if 'reunioes_semana' in work_data.columns:
            avg_meetings = work_data['reunioes_semana'].mean()
            if avg_meetings > 10:
                patterns['meeting_overload'] = f"Média de {avg_meetings:.1f} reuniões/semana"
            elif avg_meetings &lt; 2:
                patterns['isolation'] = f"Poucas reuniões ({avg_meetings:.1f}/semana)"
        
        return patterns
```

## 🔷 4. Dados Sensíveis e Privacidade

### ✅ **Anonimização de dados pessoais**:

```python
import hashlib
import uuid

class DataAnonymizer:
    def __init__(self, salt='company_salt_2026'):
        self.salt = salt
    
    def anonymize_text(self, text):
        """Anonimizar texto sensível"""
        if not isinstance(text, str):
            return text
        
        # Para nomes: substituir por hash
        if len(text.split()) &lt;= 3:  # Provavelmente um nome
            return f"USER_{hashlib.sha256((text + self.salt).encode()).hexdigest()[:8]}"
        
        return text
    
    def anonymize_email(self, email):
        """Anonimizar email mantendo domínio"""
        if '@' in email:
            local_part, domain = email.split('@')
            anonymized_local = hashlib.sha256((local_part + self.salt).encode()).hexdigest()[:6]
            return f"{anonymized_local}@{domain}"
        return email
    
    def generalize_age(self, age):
        """Generalizar idade em faixas"""
        if age &lt; 18:
            return "&lt;18"
        elif age &lt; 25:
            return "18-24"
        elif age &lt; 35:
            return "25-34"
        elif age &lt; 45:
            return "35-44"
        elif age &lt; 55:
            return "45-54"
        else:
            return "55+"
    
    def anonymize_dataframe(self, df, sensitive_columns):
        """Anonimizar DataFrame completo"""
        df_anon = df.copy()
        
        for col in sensitive_columns:
            if col in df_anon.columns:
                if col == 'email':
                    df_anon[col] = df_anon[col].apply(self.anonymize_email)
                elif col == 'idade':
                    df_anon[col] = df_anon[col].apply(self.generalize_age)
                elif df_anon[col].dtype == 'object':
                    df_anon[col] = df_anon[col].apply(self.anonymize_text)
        
        return df_anon

# Dados sensíveis de exemplo
```

```python
        if all(col in survey_data.columns for col in self.wellbeing_indicators['stress_level']):
            stress_score = (
                (1 - survey_data['prazos_apertados'].mean()) * 0.4 +
                (1 - survey_data['conflitos_equipe'].mean()) * 0.3 +
                (survey_data['carga_trabalho'].mean() &lt;= 0.7) * 0.3
            ) * 100
```
