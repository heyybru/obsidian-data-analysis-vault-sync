## 🔷 Encontrando o Inesperado nos Dados

**Detecção de anomalias é como um "sistema de alarme" para dados**.  
Você define o que é "normal" e o sistema alerta quando algo foge do padrão.

✅ **No seu contexto (RH + QA)**:

- **RH**: Detectar fraude em currículos, turnover anormal, padrões suspeitos de contratação
- **QA**: Identificar bugs críticos, performance fora do padrão, ataques de segurança

## 🔷 1. Métodos Estatísticos Básicos

### ✅ **Z-Score (para distribuição normal)**:

```python
def detect_anomalies_zscore(data, threshold=3):
    """
    Detectar anomalias usando Z-Score
    threshold: número de desvios padrão para considerar anomalia
    """
    mean = np.mean(data)
    std = np.std(data)
    
    z_scores = np.abs((data - mean) / std)
    anomalies = data[z_scores > threshold]
    anomaly_indices = np.where(z_scores > threshold)[0]
    
    return anomalies, anomaly_indices, z_scores

# Exemplo: Salários em uma empresa
salarios = np.concatenate([
    np.random.normal(5000, 500, 100),  # Normal
    np.array([10000, 12000, 15000])    # Anomalias (salários extremos)
])

anomalias, indices, z_scores = detect_anomalies_zscore(salarios, threshold=3)

print("💰 Detecção de Anomalias em Salários (Z-Score):")
print(f"Total salários: {len(salarios)}")
print(f"Média: R$ {np.mean(salarios):.2f}")
print(f"Desvio padrão: R$ {np.std(salarios):.2f}")
print(f"\n🚨 Anomalias detectadas (Z-Score > 3):")
for i, (idx, salario) in enumerate(zip(indices, anomalias)):
    print(f"  {i+1}. Índice {idx}: R$ {salario:.2f} (Z-Score: {z_scores[idx]:.2f})")

# Visualizar
plt.figure(figsize=(10, 6))
plt.scatter(range(len(salarios)), salarios, alpha=0.6, label='Dados normais')
plt.scatter(indices, anomalias, color='red', s=100, label='Anomalias')
plt.axhline(y=np.mean(salarios), color='green', linestyle='--', label='Média')
plt.axhline(y=np.mean(salarios) + 3*np.std(salarios), color='orange', 
            linestyle=':', label='Limite (+3σ)')
plt.axhline(y=np.mean(salarios) - 3*np.std(salarios), color='orange', 
            linestyle=':', label='Limite (-3σ)')
plt.xlabel('Índice')
plt.ylabel('Salário (R$)')
plt.title('Detecção de Anomalias em Salários (Z-Score)')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
```

### ✅ **IQR (Interquartile Range) - robusto a outliers**:

```python
def detect_anomalies_iqr(data, multiplier=1.5):
    """
    Detectar anomalias usando IQR (mais robusto que Z-Score)
    """
    q1 = np.percentile(data, 25)
    q3 = np.percentile(data, 75)
    iqr = q3 - q1
    
    lower_bound = q1 - multiplier * iqr
    upper_bound = q3 + multiplier * iqr
    
    anomalies = data[(data &lt; lower_bound) | (data > upper_bound)]
    anomaly_indices = np.where((data &lt; lower_bound) | (data > upper_bound))[0]
    
    return anomalies, anomaly_indices, (lower_bound, upper_bound)

# Exemplo: Tempo de resolução de bugs (QA)
tempos_resolucao = np.concatenate([
    np.random.exponential(2, 100),  # Tempos normais (2 horas em média)
    np.array([10, 12, 15, 8])       # Anomalias (tempos muito longos)
])

anomalias_iqr, indices_iqr, bounds = detect_anomalies_iqr(tempos_resolucao, multiplier=1.5)

print("\n⏱️ Detecção de Anomalias em Tempos de Resolução (IQR):")
print(f"Total tempos: {len(tempos_resolucao)}")
print(f"Q1 (25%): {np.percentile(tempos_resolucao, 25):.2f}h")
print(f"Q3 (75%): {np.percentile(tempos_resolucao, 75):.2f}h")
print(f"IQR: {np.percentile(tempos_resolucao, 75) - np.percentile(tempos_resolucao, 25):.2f}h")
print(f"Limite inferior: {bounds[0]:.2f}h")
print(f"Limite superior: {bounds[1]:.2f}h")

print(f"\n🚨 Anomalias detectadas (IQR method):")
for i, (idx, tempo) in enumerate(zip(indices_iqr, anomalias_iqr)):
    print(f"  {i+1}. Índice {idx}: {tempo:.2f}h")
```

## 🔷 2. Machine Learning para Detecção de Anomalias

### ✅ **Isolation Forest (para dados multivariados)**:

```python
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import StandardScaler
import pandas as pd

# Dados de funcionários (RH)
np.random.seed(42)
n_funcionarios = 1000

# Características normais
dados_normais = pd.DataFrame({
    'idade': np.random.normal(35, 5, n_funcionarios),
    'experiencia': np.random.normal(5, 2, n_funcionarios),
    'salario': np.random.normal(6000, 1000, n_funcionarios),
    'horas_trabalhadas': np.random.normal(160, 10, n_funcionarios)
})

# Adicionar algumas anomalias
anomalias_rh = pd.DataFrame({
    'idade': [25, 60, 22, 65],  # Idades extremas
    'experiencia': [15, 1, 20, 0.5],  # Experiência extrema
    'salario': [15000, 2000, 18000, 1500],  # Salários extremos
    'horas_trabalhadas': [300, 80, 350, 70]  # Horas extremas
})

# Combinar dados
dados_completos = pd.concat([dados_normais, anomalias_rh], ignore_index=True)

# Normalizar dados
scaler = StandardScaler()
dados_normalizados = scaler.fit_transform(dados_completos)

# Aplicar Isolation Forest
iso_forest = IsolationForest(
    contamination=0.05,  # Espera-se ~5% de anomalias
    random_state=42,
    n_estimators=100
)

# Treinar e prever
predicoes = iso_forest.fit_predict(dados_normalizados)

# -1 = anomalia, 1 = normal
dados_completos['anomalia'] = predicoes
dados_completos['score_anomalia'] = iso_forest.score_samples(dados_normalizados)

print("🌲 Detecção de Anomalias com Isolation Forest (RH):")
print(f"Total funcionários: {len(dados_completos)}")
print(f"Anomalias detectadas: {(dados_completos['anomalia'] == -1).sum()}")
print(f"Porcentagem de anomalias: {(dados_completos['anomalia'] == -1).mean()*100:.1f}%")

# Mostrar anomalias
anomalias_detectadas = dados_completos[dados_completos['anomalia'] == -1]
print("\n🚨 Funcionários Anômalos Detectados:")
for idx, row in anomalias_detectadas.iterrows():
    print(f"\n  Funcionário {idx}:")
    print(f"    Idade: {row['idade']:.0f} anos")
    print(f"    Experiência: {row['experiencia']:.1f} anos")
    print(f"    Salário: R$ {row['salario']:.2f}")
    print(f"    Horas trabalhadas: {row['horas_trabalhadas']:.0f}h")
    print(f"    Score anomalia: {row['score_anomalia']:.3f}")

# Visualizar em 2D (idade vs salário)
plt.figure(figsize=(10, 6))
cores = ['blue' if a == 1 else 'red' for a in dados_completos['anomalia']]
plt.scatter(dados_completos['idade'], dados_completos['salario'], 
            c=cores, alpha=0.6, s=50)
plt.xlabel('Idade')
plt.ylabel('Salário (R$)')
plt.title('Detecção de Anomalias em Funcionários (Isolation Forest)')
plt.grid(True, alpha=0.3)
plt.show()
```

### ✅ **One-Class SVM (para dados complexos)**:

```python
from sklearn.svm import OneClassSVM
from sklearn.decomposition import PCA

# Dados de logs de sistema (QA)
np.random.seed(42)
n_logs = 500

# Logs normais
logs_normais = pd.DataFrame({
    'tempo_resposta': np.random.exponential(0.5, n_logs),  # 0.5s em média
    'cpu_uso': np.random.beta(2, 5, n_logs) * 100,  # 0-100%
    'memoria_uso': np.random.beta(3, 4, n_logs) * 100,
    'erros': np.random.poisson(0.1, n_logs)  # Poucos erros
})

# Adicionar anomalias (ataques/falhas)
logs_anomalos = pd.DataFrame({
    'tempo_resposta': np.random.exponential(5, 20),  # Muito lento
    'cpu_uso': np.random.uniform(90, 100, 20),  # CPU no limite
    'memoria_uso': np.random.uniform(95, 100, 20),  # Memória no limite
    'erros': np.random.poisson(10, 20)  # Muitos erros
})

# Combinar
logs_completos = pd.concat([logs_normais, logs_anomalos], ignore_index=True)

# Normalizar
scaler_logs = StandardScaler()
logs_normalizados = scaler_logs.fit_transform(logs_completos)

# One-Class SVM
oc_svm = OneClassSVM(
    nu=0.05,  # Proporção esperada de outliers
    kernel='rbf',  # Kernel para dados não-lineares
    gamma='auto'
)

# Treinar apenas com dados normais (primeiros 500)
oc_svm.fit(logs_normalizados[:500])

# Prever para todos
predicoes_svm = oc_svm.predict(logs_normalizados)
logs_completos['anomalia_svm'] = predicoes_svm

print("🛡️ Detecção de Anomalias com One-Class SVM (QA):")
print(f"Total logs: {len(logs_completos)}")
print(f"Logs normais usados para treino: 500")
print(f"Anomalias detectadas: {(logs_completos['anomalia_svm'] == -1).sum()}")
print(f"Porcentagem de anomalias: {(logs_completos['anomalia_svm'] == -1).mean()*100:.1f}%")

# PCA para visualização
pca = PCA(n_components=2)
logs_pca = pca.fit_transform(logs_normalizados)

# Visualizar
plt.figure(figsize=(12, 5))

# Plot 1: Features originais
plt.subplot(1, 2, 1)
cores_svm = ['blue' if a == 1 else 'red' for a in logs_completos['anomalia_svm']]
plt.scatter(logs_completos['tempo_resposta'], logs_completos['cpu_uso'], 
            c=cores_svm, alpha=0.6, s=50)
plt.xlabel('Tempo de Resposta (s)')
plt.ylabel('Uso de CPU (%)')
plt.title('Tempo vs CPU (One-Class SVM)')
plt.grid(True, alpha=0.3)

# Plot 2: PCA
plt.subplot(1, 2, 2)
plt.scatter(logs_pca[:, 0], logs_pca[:, 1], c=cores_svm, alpha=0.6, s=50)
plt.xlabel('Componente Principal 1')
plt.ylabel('Componente Principal 2')
plt.title('Visualização PCA (One-Class SVM)')
plt.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
```

## 🔷 3. Detecção de Anomalias em Séries Temporais

### ✅ **STL Decomposition + IQR**:

```python
from statsmodels.tsa.seasonal import STL
import pandas as pd

# Criar série temporal de bugs reportados
np.random.seed(42)
n_dias = 365
datas = pd.date_range('2025-01-01', periods=n_dias, freq='D')

# Padrão normal: sazonalidade semanal + tendência
tendencia = np.linspace(20, 40, n_dias)  # Aumento gradual
sazonalidade = 10 * np.sin(2 * np.pi * np.arange(n_dias) / 7)  # Semanal
ruido = np.random.normal(0, 3, n_dias)

bugs_diarios = tendencia + sazonalidade + ruido

# Adicionar anomalias
indices_anomalias = [50, 100, 150, 200, 250, 300]
bugs_diarios[indices_anomalias] = [80, 85, 90, 75, 95, 100]  # Picos anormais

# Criar série temporal
serie_bugs = pd.Series(bugs_diarios, index=datas)

# Decomposição STL
stl = STL(serie_bugs, period=7, seasonal=7)  # Sazonalidade semanal
resultado = stl.fit()

# Extrair resíduos
residuos = resultado.resid

# Detectar anomalias nos resíduos
anomalias_residuos, indices_residuos, bounds_residuos = detect_anomalies_iqr(
    residuos.values, multiplier=3
)

print("📈 Detecção de Anomalias em Séries Temporais (STL + IQR):")
print(f"Período analisado: {datas[0].date()} a {datas[-1].date()}")
print(f"Total dias: {len(serie_bugs)}")
print(f"Anomalias detectadas: {len(anomalias_residuos)}")
print(f"Média de bugs: {serie_bugs.mean():.1f}/dia")
print(f"Desvio padrão: {serie_bugs.std():.1f}")

# Visualizar decomposição
fig, axes = plt.subplots(4, 1, figsize=(12, 10))

# Série original
axes[0].plot(serie_bugs, label='Bugs Diários')
axes[0].scatter(serie_bugs.index[indices_anomalias], 
                serie_bugs.iloc[indices_anomalias], 
                color='red', s=50, label='Anomalias inseridas')
axes[0].set_title('Bugs Reportados por Dia')
axes[0].legend()
axes[0].grid(True, alpha=0.3)

# Tendência
axes[1].plot(resultado.trend, label='Tendência', color='orange')
axes[1].set_title('Tendência')
axes[1].grid(True, alpha=0.3)

# Sazonalidade
axes[2].plot(resultado.seasonal, label='Sazonalidade', color='green')
axes[2].set_title('Sazonalidade (Semanal)')
axes[2].grid(True, alpha=0.3)

# Resíduos com anomalias
axes[3].plot(residuos, label='Resíduos', color='purple')
```
