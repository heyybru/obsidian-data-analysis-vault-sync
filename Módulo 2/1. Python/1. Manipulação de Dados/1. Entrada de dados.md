## 🔷 6. Tratamento de Erros na Entrada

### ✅ **Try-Except para robustez**:

```python
def ler_arquivo_seguro(caminho):
    try:
        with open(caminho, 'r', encoding='utf-8') as arquivo:
            return arquivo.read()
    except FileNotFoundError:
        print(f"❌ Arquivo {caminho} não encontrado")
        return None
    except PermissionError:
        print(f"❌ Sem permissão para ler {caminho}")
        return None
    except UnicodeDecodeError:
        print(f"❌ Problema de codificação em {caminho}")
        return None

# Uso
conteudo = ler_arquivo_seguro('dados.txt')
if conteudo:
    print("✅ Arquivo lido com sucesso")
```

## 🔷 7. Exemplos Práticos (RH + QA)

### ✅ **RH: Sistema de Cadastro de Candidatos**:

```python
import pandas as pd
import os

class SistemaRH:
    def __init__(self):
        self.arquivo = 'candidatos.csv'
        if not os.path.exists(self.arquivo):
            pd.DataFrame(columns=['nome', 'idade', 'experiencia', 'salario']).to_csv(self.arquivo, index=False)
    
    def cadastrar_candidato(self):
        print("=== Cadastro de Candidato ===")
        candidato = {
            'nome': input("Nome: "),
            'idade': int(input("Idade: ")),
            'experiencia': float(input("Anos de experiência: ")),
            'salario': float(input("Pretensão salarial: "))
        }
        
        # Salvar no CSV
        df = pd.read_csv(self.arquivo)
        df = pd.concat([df, pd.DataFrame([candidato])], ignore_index=True)
        df.to_csv(self.arquivo, index=False)
        print("✅ Candidato cadastrado!")
    
    def analisar_candidatos(self):
        df = pd.read_csv(self.arquivo)
        print(f"\n📊 Total de candidatos: {len(df)}")
        print(f"Média salarial: R$ {df['salario'].mean():.2f}")
        print(f"Média de experiência: {df['experiencia'].mean():.1f} anos")

# Uso
rh = SistemaRH()
rh.cadastrar_candidato()
rh.analisar_candidatos()
```

### ✅ **QA: Sistema de Relatórios de Bugs**:

```python
class SistemaQA:
    def importar_bugs_csv(self, arquivo_csv):
        try:
            df = pd.read_csv(arquivo_csv)
            print(f"✅ Importados {len(df)} bugs do CSV")
            
            # Análise básica
            print("\n📊 Estatísticas:")
            print(df['severidade'].value_counts())
            
            # Bugs por módulo
            bugs_modulo = df.groupby('modulo').size().sort_values(ascending=False)
            print(f"\n🐛 Bugs por módulo:\n{bugs_modulo}")
            
            return df
            
        except FileNotFoundError:
            print(f"❌ Arquivo {arquivo_csv} não encontrado")
            return None
    
    def gerar_relatorio(self, df_bugs):
        if df_bugs is None:
            return
        
        # Salvar relatório
        relatorio = df_bugs.groupby(['modulo', 'severidade']).size().unstack(fill_value=0)
        relatorio.to_csv('relatorio_bugs.csv')
        print("✅ Relatório salvo como 'relatorio_bugs.csv'")

# Uso
qa = SistemaQA()
df_bugs = qa.importar_bugs_csv('bugs.csv')
qa.gerar_relatorio(df_bugs)
```

def validar_candidato(dados):
    erros = []
    
    if not dados.get('nome'):
        erros.append("Nome obrigatório")
    
    if dados.get('idade', 0) < 18:
        erros.append("Idade deve ser >= 18")
    
    if dados.get('salario', 0) <= 0:
        erros.append("Salário deve ser positivo")
    
    return erros

# Uso
dados = {'nome': 'Ana', 'idade': 16, 'salario': 3000}

    print(f"Erros de validação: {erros}")
