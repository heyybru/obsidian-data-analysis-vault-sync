## 🔷 Statsmodels: Estatística Inferencial em Python

**Statsmodels é como um "laboratório de estatística" completo para análise inferencial**.  
Enquanto SciPy faz testes básicos, Statsmodels fornece **modelos estatísticos completos** com diagnósticos detalhados.

✅ **No seu contexto (RH + QA)**:

- **RH**: Regressão linear para prever salários, ANOVA para comparar departamentos
- **QA**: Modelos de sobrevivência para tempo até bug, séries temporais para previsão

---

## 🔷 O que é Statsmodels?

### ✅ **Características principais**:

- **Modelos estatísticos clássicos**: Regressão, ANOVA, séries temporais

- **Foco em inferência**: Testes de hipóteses, significância estatística

## 🔷 Regressão Linear com Statsmodels

### ✅ **Regressão simples (RH: Salário vs Experiência)**:

```python
import pandas as pd
import numpy as np
import statsmodels.api as sm
import matplotlib.pyplot as plt

# Dados simulados
np.random.seed(42)
n = 100
experiencia = np.random.normal(5, 2, n)
salario = 3000 + 500 * experiencia + np.random.normal(0, 300, n)

# Criar DataFrame
df = pd.DataFrame({'experiencia': experiencia, 'salario': salario})

# Adicionar constante para intercepto
X = sm.add_constant(df['experiencia'])  # Adiciona coluna de 1s
y = df['salario']

# Ajustar modelo
modelo = sm.OLS(y, X).fit()

# Resumo completo
print(modelo.summary())
```


## 🔷 Regressão Múltipla (RH)

### ✅ **Prever salário com múltiplas variáveis**:

```python
import statsmodels.formula.api as smf

# Dados mais complexos
np.random.seed(42)
n = 200

df = pd.DataFrame({
    'experiencia': np.random.normal(5, 2, n),
    'idade': np.random.normal(30, 5, n),
    'certificacoes': np.random.poisson(2, n),
    'departamento': np.random.choice(['TI', 'RH', 'Vendas'], n),
    'genero': np.random.choice(['M', 'F'], n)
})

# Salário baseado nas variáveis + ruído
df['salario'] = (
    3000 + 
    400 * df['experiencia'] + 
    50 * df['idade'] + 
    200 * df['certificacoes'] +
    np.where(df['departamento'] == 'TI', 1000, 
             np.where(df['departamento'] == 'Vendas', 500, 0)) +
    np.where(df['genero'] == 'M', 200, 0) +
    np.random.normal(0, 300, n)
)

# Modelo com fórmula (similar ao R)
modelo = smf.ols(
    'salario ~ experiencia + idade + certificacoes + C(departamento) + C(genero)',
    data=df
).fit()

print(modelo.summary())
```

## 🔷 ANOVA (Análise de Variância)

### ✅ **Comparar salários entre departamentos**:

```python
from statsmodels.stats.anova import anova_lm

# ANOVA unidirecional
modelo_anova = smf.ols('salario ~ C(departamento)', data=df).fit()
anova_table = anova_lm(modelo_anova)

print("📊 ANOVA - Salários por Departamento")
print(anova_table)
print("\n")

# Teste post-hoc (Tukey HSD)
from statsmodels.stats.multicomp import pairwise_tukeyhsd

tukey = pairwise_tukeyhsd(
    endog=df['salario'],
    groups=df['departamento'],
    alpha=0.05
)

print("🔍 Teste de Tukey (comparações múltiplas):")
print(tukey.summary())

# Visualizar
fig = tukey.plot_simultaneous()
plt.title('Comparação de Salários entre Departamentos')
plt.show()
```

### ✅ **ANOVA bidirecional (com interação)**:

```python
# Salário por departamento e gênero
modelo_anova2 = smf.ols(
    'salario ~ C(departamento) * C(genero)',  # * inclui interação
    data=df
).fit()

anova_table2 = anova_lm(modelo_anova2, typ=2)
print("📊 ANOVA Bidirecional com Interação")
print(anova_table2)

# Verificar interação significativa
if anova_table2.loc['C(departamento):C(genero)', 'PR(>F)'] < 0.05:
    print("\n⚠️ Interação significativa: Efeito do departamento depende do gênero")
else:
    print("\n✅ Sem interação significativa: Efeitos são independentes")
```

## 🔷 Modelos Logísticos (Classificação)

### ✅ **Regressão Logística (RH: Prever contratação)**:

```python
# Dados binários: 1=contratado, 0=não contratado
np.random.seed(42)
n = 300

df_logistic = pd.DataFrame({
    'experiencia': np.random.normal(5, 2, n),
    'idade': np.random.normal(30, 5, n),
    'entrevista_score': np.random.uniform(0, 10, n),
    'tem_ingles': np.random.binomial(1, 0.7, n)
})

# Probabilidade de contratação (função logística)
log_odds = (
    -3 + 
    0.5 * df_logistic['experiencia'] +
    0.1 * df_logistic['idade'] +
    0.3 * df_logistic['entrevista_score'] +
    0.8 * df_logistic['tem_ingles']
)
probabilidade = 1 / (1 + np.exp(-log_odds))
df_logistic['contratado'] = np.random.binomial(1, probabilidade)

# Modelo logístico
modelo_logistic = smf.logit(
    'contratado ~ experiencia + idade + entrevista_score + tem_ingles',
    data=df_logistic
).fit()

print(modelo_logistic.summary())

# Interpretar odds ratios
odds_ratios = np.exp(modelo_logistic.params)
print("\n📈 Odds Ratios (razão de chances):")
for var, odds in odds_ratios.items():
    print(f"{var}: {odds:.3f}")
    if odds > 1:
        print(f"  → Aumenta chance em {(odds-1)*100:.1f}%")
    else:
        print(f"  → Reduz chance em {(1-odds)*100:.1f}%")
```

### ✅ **Curva ROC e AUC**:

```python
from sklearn.metrics import roc_curve, auc

# Probabilidades previstas
probabilidades = modelo_logistic.predict(df_logistic)

# Curva ROC
fpr, tpr, thresholds = roc_curve(df_logistic['contratado'], probabilidades)
roc_auc = auc(fpr, tpr)

# Plot
plt.figure(figsize=(8, 6))
plt.plot(fpr, tpr, color='darkorange', lw=2, 
         label=f'Curva ROC (AUC = {roc_auc:.3f})')
plt.plot([0, 1], [0, 1], color='navy', lw=2, linestyle='--')
plt.xlim([0.0, 1.0])
plt.ylim([0.0, 1.05])
plt.xlabel('Taxa de Falsos Positivos')
plt.ylabel('Taxa de Verdadeiros Positivos')
plt.title('Curva ROC - Modelo de Contratação')
plt.legend(loc="lower right")
plt.grid(True, alpha=0.3)
plt.show()
```

## 🔷 Séries Temporais (QA)

### ✅ **Modelo ARIMA para prever bugs**:

```python
import statsmodels.tsa.api as tsa
from statsmodels.tsa.stattools import adfuller

# Dados temporais de bugs
np.random.seed(42)
n_months = 60
meses = pd.date_range('2021-01-01', periods=n_months, freq='M')

# Série temporal com tendência e sazonalidade
tendencia = np.linspace(20, 80, n_months)
sazonalidade = 10 * np.sin(2 * np.pi * np.arange(n_months) / 12)
ruido = np.random.normal(0, 5, n_months)
bugs = tendencia + sazonalidade + ruido

df_ts = pd.DataFrame({'bugs': bugs}, index=meses)

# Teste de estacionariedade (Dickey-Fuller)
result_adf = adfuller(df_ts['bugs'])
print(f"📊 Teste Dickey-Fuller:")
print(f"  Estatística ADF: {result_adf[0]:.3f}")
print(f"  P-valor: {result_adf[1]:.3f}")
print(f"  {'✅ Estacionária' if result_adf[1] < 0.05 else '❌ Não estacionária'}")

# Diferenciar se necessário
if result_adf[1] > 0.05:
    df_ts['bugs_diff'] = df_ts['bugs'].diff().dropna()
    print("  Aplicada diferenciação para tornar estacionária")

# ACF e PACF para identificar ordem
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(10, 8))
tsa.graphics.plot_acf(df_ts['bugs'].dropna(), lags=20, ax=ax1)
tsa.graphics.plot_pacf(df_ts['bugs'].dropna(), lags=20, ax=ax2)
plt.show()

# Modelo ARIMA
modelo_arima = tsa.ARIMA(
    df_ts['bugs'],
    order=(1, 1, 1),  # (p, d, q)
    seasonal_order=(1, 1, 1, 12)  # Sazonalidade anual
).fit()

print(modelo_arima.summary())

# Previsão
forecast_steps = 12
forecast = modelo_arima.forecast(steps=forecast_steps)
forecast_index = pd.date_range(df_ts.index[-1], periods=forecast_steps+1, freq='M')[1:]

# Plot
plt.figure(figsize=(12, 6))
plt.plot(df_ts.index, df_ts['bugs'], label='Dados históricos')
plt.plot(forecast_index, forecast, label='Previsão', color='red', linestyle='--')
plt.fill_between(forecast_index, 
                 forecast - 1.96*modelo_arima.bse, 
                 forecast + 1.96*modelo_arima.bse,
                 color='red', alpha=0.2, label='Intervalo 95%')
plt.title('Previsão de Bugs - Modelo ARIMA')
plt.xlabel('Data')
plt.ylabel('Número de Bugs')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
```
