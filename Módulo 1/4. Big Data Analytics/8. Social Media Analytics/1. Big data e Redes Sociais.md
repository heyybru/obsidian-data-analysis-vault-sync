### ✅ **Pipeline de processamento**:

```python
# Arquitetura Lambda para redes sociais
import pandas as pd
import numpy as np
from datetime import datetime, timedelta

class SocialMediaBigDataPipeline:
    def __init__(self):
        self.batch_size = 1000000  # 1 milhão de registros por batch
        self.realtime_window = 300  # 5 minutos para streaming
        
    def ingest_data(self, platform, data_type):
        """Ingestão de dados de diferentes plataformas"""
        ingestion_strategies = {
            'twitter': self._ingest_twitter,
            'facebook': self._ingest_facebook,
            'instagram': self._ingest_instagram,
            'linkedin': self._ingest_linkedin
        }
        
        return ingestion_strategies.get(platform, self._ingest_generic)(data_type)
    
    def _ingest_twitter(self, data_type):
        """Ingestão específica do Twitter/X"""
        # API v2 permite 10 milhões de tweets/mês
        # Streaming API para tempo real
        return {
            'rate_limit': '10M/month',
            'realtime': True,
            'fields': ['id', 'text', 'created_at', 'author_id', 
                      'retweet_count', 'like_count', 'reply_count']
        }
    
    def process_batch(self, data):
        """Processamento batch (Hadoop/Spark)"""
        # Exemplo: Processar 1 dia de dados
        daily_stats = {
            'total_posts': len(data),
            'unique_users': data['user_id'].nunique(),
            'avg_engagement': data['engagement'].mean(),
            'peak_hour': data['hour'].mode()[0] if len(data) > 0 else None
        }
        return daily_stats
    
    def process_stream(self, data_stream):
        """Processamento streaming (Flink/Kafka)"""
        # Detectar tendências em tempo real
        trends = {
            'hot_topics': self._detect_trending_topics(data_stream),
            'sentiment_shifts': self._detect_sentiment_changes(data_stream),
            'viral_content': self._identify_viral_posts(data_stream)
        }
        return trends
```

## 🔷 Análise de Redes Sociais em Escala

### ✅ **Grafo social com NetworkX**:

```python
import networkx as nx
import pandas as pd
import matplotlib.pyplot as plt
from collections import Counter

# Simular rede social com 10.000 usuários
np.random.seed(42)
n_users = 10000

# Criar grafo
G = nx.Graph()

# Adicionar nós (usuários)
for i in range(n_users):
    G.add_node(i, 
               followers=np.random.poisson(500),
               activity=np.random.exponential(10))

# Adicionar arestas (conexões)
# Lei de Pareto: poucos usuários têm muitas conexões
for i in range(n_users):
    # Número de conexões segue distribuição power-law
    n_connections = np.random.zipf(1.5)
    n_connections = min(n_connections, 1000)  # Limite máximo
    
    # Conectar com outros usuários (preferência por usuários populares)
    popular_users = sorted(range(n_users), 
                          key=lambda x: G.nodes[x]['followers'], 
                          reverse=True)[:1000]
    
    connections = np.random.choice(popular_users, 
                                  size=min(n_connections, 1000), 
                                  replace=False)
    
    for conn in connections:
        if i != conn:
            G.add_edge(i, conn)

# Métricas da rede
print("📊 Análise da Rede Social:")
print(f"Número de nós: {G.number_of_nodes():,}")
print(f"Número de arestas: {G.number_of_edges():,}")
print(f"Densidade: {nx.density(G):.6f}")

# Grau médio
degrees = [d for n, d in G.degree()]
print(f"Grau médio: {np.mean(degrees):.1f}")
print(f"Grau máximo: {max(degrees):,}")

# Coeficiente de clustering
clustering = nx.average_clustering(G)
print(f"Coeficiente de clustering: {clustering:.3f}")

# Diâmetro (aproximado para redes grandes)
if G.number_of_nodes() &lt;= 1000:
    print(f"Diâmetro: {nx.diameter(G)}")
else:
    # Para redes grandes, estimar
    print("Diâmetro: (muito grande para calcular)")

# Encontrar influenciadores (nós com maior centralidade)
degree_centrality = nx.degree_centrality(G)
top_influencers = sorted(degree_centrality.items(), 
                        key=lambda x: x[1], 
                        reverse=True)[:10]

print("\n🏆 Top 10 Influenciadores:")
for user, centrality in top_influencers:
    followers = G.nodes[user]['followers']
    print(f"  Usuário {user}: Centralidade={centrality:.4f}, Seguidores={followers:,}")
```

### ✅ **Detecção de comunidades**:

```python
from community import community_louvain  # python-louvain

# Detectar comunidades
partition = community_louvain.best_partition(G)

# Análise das comunidades
communities = {}
for node, comm_id in partition.items():
    if comm_id not in communities:
        communities[comm_id] = []
    communities[comm_id].append(node)

print(f"\n🔍 Detecção de Comunidades:")
print(f"Número de comunidades detectadas: {len(communities)}")

# Tamanho das comunidades
community_sizes = [len(members) for members in communities.values()]
print(f"Maior comunidade: {max(community_sizes):,} membros")
print(f"Menor comunidade: {min(community_sizes):,} membros")
print(f"Média: {np.mean(community_sizes):.1f} membros")

# Modularidade (qualidade da detecção)
modularity = community_louvain.modularity(partition, G)
print(f"Modularidade: {modularity:.3f} (quanto maior, melhor)")

# Analisar comunidades específicas
print("\n📈 Análise das 5 maiores comunidades:")
sorted_communities = sorted(communities.items(), 
                           key=lambda x: len(x[1]),

    # Características da comunidade

    # Densidade interna
```

```python
from textblob import TextBlob

    .config("spark.executor.memory", "8g") \
    .config("spark.driver.memory", "4g") \
    .getOrCreate()

# Em produção, isso viria de HDFS/S3
data = [
    ("post1", "Adorei o novo produto! Muito útil.", "2026-01-01 10:00", "twitter"),
    ("post2", "Não gostei da atualização, está muito lento.", "2026-01-01 10:05", "facebook"),
    ("post3", "O suporte foi excelente, resolveram rápido.", "2026-01-01 10:10", "instagram"),
    # ... 9.999.997 posts adicionais
]

# Criar DataFrame Spark
columns = ["post_id", "text", "timestamp", "platform"]
df = spark.createDataFrame(data, columns)

# Função UDF para análise de sentimento
def analyze_sentiment(text):
    try:
        blob = TextBlob(text)
        polarity = blob.sentiment.polarity
        
        if polarity > 0.1:
            return "POSITIVE"
        elif polarity &lt; -0.1:
            return "NEGATIVE"
        else:
            return "NEUTRAL"
    except:
        return "NEUTRAL"

# Registrar UDF
sentiment_udf = udf(analyze_sentiment, StringType())

# Aplicar análise
df_with_sentiment = df.withColumn("sentiment", sentiment_udf(col("text")))

# Agregar resultados
sentiment_stats = df_with_sentiment.groupBy("sentiment", "platform") \
    .count() \
    .orderBy("platform", "sentiment")

# Mostrar resultados
print("📊 Análise de Sentimento em Escala:")
sentiment_stats.show()

# Calcular porcentagens
total_posts = df_with_sentiment.count()
positive_count = df_with_sentiment.filter(col("sentiment") == "POSITIVE").count()
negative_count = df_with_sentiment.filter(col("sentiment") == "NEGATIVE").count()

print(f"\n📈 Estatísticas Gerais:")
print(f"Total posts analisados: {total_posts:,}")
print(f"Positivos: {positive_count:,} ({positive_count/total_posts*100:.1f}%)")
print(f"Negativos: {negative_count:,} ({negative_count/total_posts*100:.1f}%)")
print(f"Neutros: {total_posts - positive_count - negative_count:,} ({(total_posts - positive_count - negative_count)/total_posts*100:.1f}%)")

# Análise temporal (por hora)
df_with_hour = df_with_sentiment.withColumn("hour", F.hour(col("timestamp")))
hourly_sentiment = df_with_hour.groupBy("hour", "sentiment") \
    .count() \
    .orderBy("hour", "sentiment")

print("\n⏰ Análise por Hora:")
hourly_sentiment.show(24)  # Mostrar 24 horas

# Parar Spark
spark.stop()
```

## 🔷 Detecção de Tendências em Tempo Real

### ✅ **Sistema de alerta para crises**:

```python
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
from collections import deque
import warnings

class SocialMediaCrisisDetector:
    def __init__(self, threshold=3.0, window_size=60):
        """
        threshold: número de desvios padrão para alerta
        window_size: tamanho da janela em minutos
        """
        self.threshold = threshold
        self.window_size = window_size
        self.metrics_history = deque(maxlen=window_size)
        self.alerts = []
        
    def process_stream(self, data_point):
        """Processar ponto de dados em tempo real"""
        current_time = datetime.now()
        
        # Extrair métricas
        metrics = {
            'timestamp': current_time,
            'mention_count': data_point.get('mentions', 0),
            'negative_sentiment': data_point.get('negative_ratio', 0),
            'engagement_rate': data_point.get('engagement', 0)
        }
        
        # Adicionar ao histórico
        self.metrics_history.append(metrics)
        
        # Detectar anomalias se tivermos dados suficientes
        if len(self.metrics_history) >= 10:  # Mínimo 10 pontos
            self._detect_anomalies(metrics)
        
        return metrics
    
    def _detect_anomalies(self, current_metrics):
        """Detectar anomalias usando estatística"""
        # Converter histórico para DataFrame
        history_df = pd.DataFrame(list(self.metrics_history))
        
        # Calcular estatísticas
        for metric in ['mention_count', 'negative_sentiment', 'engagement_rate']:
            if metric in history_df.columns:
                values = history_df[metric].values
                
                if len(values) >= 2:
                    mean = np.mean(values[:-1])  # Média excluindo ponto atual
                    std = np.std(values[:-1]) if len(values) > 2 else 0.1
                    
                    current_value = current_metrics[metric]
                    
                    # Evitar divisão por zero
                    if std > 0:
                        z_score = abs(current_value - mean) / std
                        
                        # Verificar se é anomalia
                        if z_score > self.threshold:
                            alert = {
                                'timestamp': current_metrics['timestamp'],
                                'metric': metric,
                                'value': current_value,
                                'mean': mean,
                                'std': std,
                                'z_score': z_score,
                                'severity': self._calculate_severity(z_score)
                            }
                            
                            self.alerts.append(alert)
                            self._trigger_alert(alert)
    
    def _calculate_severity(self, z_score):
        """Calcular severidade baseada no z-score"""
        if z_score > 5:
            return 'CRITICAL'
        elif z_score > 3:
            return 'HIGH'
        elif z_score > 2:
            return 'MEDIUM'
        else:
            return 'LOW'
    
    def _trigger_alert(self, alert):
        """Disparar alerta"""
        print(f"\n🚨 ALERTA DETECTADO!")
        print(f"  Hora: {alert['timestamp'].strftime('%Y-%m-%d %H:%M:%S')}")
        print(f"  Métrica: {alert['metric']}")
        print(f"  Valor: {alert['value']:.2f} (média: {alert['mean']:.2f})")
        print(f"  Z-score: {alert['z_score']:.2f}")
        print(f"  Severidade: {alert['severity']}")
        
        # Ações recomendadas baseadas na métrica
        recommendations = {
            'mention_count': "Investigar aumento súbito de menções",
            'negative_sentiment': "Analisar reclamações recentes",
            'engagement_rate': "Verificar conteúdo viral negativo"
        }
        
        if alert['metric'] in recommendations:
            print(f"  💡 Recomendação: {recommendations[alert['metric']]}")
    
    def get_summary(self):
        """Resumo das detecções"""
        if not self.alerts:
            return "✅ Nenhum alerta detectado"
        
        df_alerts = pd.DataFrame(self.alerts)
        
        summary = f"📊 Resumo de Alertas (últimas {self.window_size} minutos):\n"
        summary += f"Total alertas: {len(self.alerts)}\n"
        
        if len(self.alerts) > 0:
            by_severity = df_alerts['severity'].value_counts()
            for severity, count in by_severity.items():
                summary += f"{severity}: {count} alertas\n"
            
            # Métrica mais problemática
            problematic_metric = df_alerts['metric'].value_counts().idxmax()
            summary += f"\n⚠️ Métrica mais problemática: {problematic_metric}"
        
        return summary
```
