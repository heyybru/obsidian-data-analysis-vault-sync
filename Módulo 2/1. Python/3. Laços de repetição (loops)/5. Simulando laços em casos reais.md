## 🔷 1) **RH: Processamento de folha de pagamento**

### ✅ **Calcular salários com descontos e bônus**:

```python
funcionarios = [
    {"nome": "Ana", "salario_base": 5000, "horas_extras": 10, "faltas": 2},
    {"nome": "Bruno", "salario_base": 6000, "horas_extras": 5, "faltas": 0},
    {"nome": "Carla", "salario_base": 4500, "horas_extras": 15, "faltas": 1},
    {"nome": "Daniel", "salario_base": 5500, "horas_extras": 0, "faltas": 3},
]

print("💰 PROCESSAMENTO DE FOLHA DE PAGAMENTO")
print("=" * 60)

for func in funcionarios:
    # Cálculos
    valor_hora_extra = (func["salario_base"] / 220) * 1.5 * func["horas_extras"]
    desconto_faltas = (func["salario_base"] / 30) * func["faltas"]
    salario_liquido = func["salario_base"] + valor_hora_extra - desconto_faltas
    
    # Output formatado
    print(f"\n👤 {func['nome']}:")
    print(f"  Salário base: R$ {func['salario_base']:.2f}")
    print(f"  Horas extras ({func['horas_extras']}h): +R$ {valor_hora_extra:.2f}")
    print(f"  Faltas ({func['faltas']} dias): -R$ {desconto_faltas:.2f}")
    print(f"  {'─' * 30}")
    print(f"  SALÁRIO LÍQUIDO: R$ {salario_liquido:.2f}")

# Totais
total_pagamento = sum(
    func["salario_base"] + 
    (func["salario_base"] / 220 * 1.5 * func["horas_extras"]) - 
    (func["salario_base"] / 30 * func["faltas"])
    for func in funcionarios
)

print(f"\n📊 TOTAL DA FOLHA: R$ {total_pagamento:.2f}")
```

## 🔷 2) **QA: Execução de suite de testes**

### ✅ **Simular execução de testes com resultados aleatórios**:

```python
import random
import time

testes = [
    {"nome": "Login", "tempo_estimado": 2, "critico": True},
    {"nome": "Cadastro", "tempo_estimado": 3, "critico": True},
    {"nome": "Pagamento", "tempo_estimado": 5, "critico": True},
    {"nome": "Perfil", "tempo_estimado": 1, "critico": False},
    {"nome": "Busca", "tempo_estimado": 2, "critico": False},
]

print("🚀 EXECUÇÃO DE SUITE DE TESTES")
print("=" * 50)

resultados = []
tempo_total = 0

for i, teste in enumerate(testes, 1):
    print(f"\n[{i}/{len(testes)}] Executando: {teste['nome']}")
    print(f"  Tempo estimado: {teste['tempo_estimado']}s | Crítico: {'✅' if teste['critico'] else '⚠️'}")
    
    # Simular execução
    time.sleep(0.5)  # Pausa para realismo
    
    # Resultado aleatório (90% de sucesso para críticos, 70% para não críticos)
    sucesso_prob = 0.9 if teste["critico"] else 0.7
    passou = random.random() < sucesso_prob
    
    # Tempo real (variação de ±20%)
    tempo_real = teste["tempo_estimado"] * random.uniform(0.8, 1.2)
    tempo_total += tempo_real
    
    resultados.append({
        "teste": teste["nome"],
        "passou": passou,
        "tempo_real": tempo_real,
        "critico": teste["critico"]
    })
    
    print(f"  Resultado: {'✅ PASSOU' if passou else '❌ FALHOU'}")
    print(f"  Tempo real: {tempo_real:.1f}s")

# Relatório final
print("\n" + "=" * 50)
print("📊 RELATÓRIO FINAL")

passaram = sum(1 for r in resultados if r["passou"])
falharam = len(resultados) - passaram
criticos_falharam = sum(1 for r in resultados if r["critico"] and not r["passou"])

print(f"\nTotal testes: {len(resultados)}")
print(f"Passaram: {passaram} ({passaram/len(resultados)*100:.1f}%)")
print(f"Falharam: {falharam} ({falharam/len(resultados)*100:.1f}%)")
print(f"Tempo total: {tempo_total:.1f}s")

if criticos_falharam > 0:
    print(f"\n⚠️  ALERTA: {criticos_falharam} teste(s) crítico(s) falharam!")
    for r in resultados:
        if r["critico"] and not r["passou"]:
            print(f"  ❌ {r['teste']}")
```

## 🔷 3) **RH: Triagem automática de currículos**

### ✅ **Filtrar candidatos por múltiplos critérios**:

```python
candidatos = [
    {"nome": "Ana", "idade": 25, "experiencia": 3, "formacao": "Superior", "pretensao": 5000},
    {"nome": "Bruno", "idade": 17, "experiencia": 1, "formacao": "Técnico", "pretensao": 3000},
    {"nome": "Carla", "idade": 30, "experiencia": 5, "formacao": "Mestrado", "pretensao": 8000},
    {"nome": "Daniel", "idade": 22, "experiencia": 0, "formacao": "Superior", "pretensao": 4000},
    {"nome": "Eduardo", "idade": 28, "experiencia": 4, "formacao": "Superior", "pretensao": 6000},
]

print("🎯 TRIAGEM AUTOMÁTICA DE CURRÍCULOS")
print("=" * 60)

# Critérios da vaga
CRITERIOS = {
    "idade_minima": 18,
    "experiencia_minima": 2,
    "formacao_minima": "Superior",
    "pretensao_maxima": 7000
}

print(f"Critérios da vaga:")
for criterio, valor in CRITERIOS.items():
    print(f"  {criterio}: {valor}")

print("\n" + "=" * 60)
print("📋 RESULTADO DA TRIAGEM:")

aprovados = []
reprovados = []

for cand in candidatos:
    print(f"\n👤 Analisando {cand['nome']}:")
    
    # Verificar cada critério
    criterios_ok = []
    
    idade_ok = cand["idade"] >= CRITERIOS["idade_minima"]
    criterios_ok.append(("Idade ≥ 18", idade_ok))
    
    exp_ok = cand["experiencia"] >= CRITERIOS["experiencia_minima"]
    criterios_ok.append(("Experiência ≥ 2 anos", exp_ok))
    
    # Comparar formação (considerando hierarquia)
    hierarquia_formacao = {"Técnico": 1, "Superior": 2, "Mestrado": 3, "Doutorado": 4}
    formacao_cand = hierarquia_formacao.get(cand["formacao"], 0)
    formacao_min = hierarquia_formacao.get(CRITERIOS["formacao_minima"], 0)
    formacao_ok = formacao_cand >= formacao_min
    criterios_ok.append((f"Formação ≥ {CRITERIOS['formacao_minima']}", formacao_ok))
    
    pretensao_ok = cand["pretensao"] <= CRITERIOS["pretensao_maxima"]
    criterios_ok.append((f"Pretensão ≤ R$ {CRITERIOS['pretensao_maxima']}", pretensao_ok))
    
    # Mostrar resultado de cada critério
    for criterio, ok in criterios_ok:
        print(f"  {'✅' if ok else '❌'} {criterio}")
    
    # Decisão final
    todos_ok = all(ok for _, ok in criterios_ok)
    
    if todos_ok:
        aprovados.append(cand["nome"])
        print(f"  🎉 RESULTADO: ✅ APROVADO")
    else:
        reprovados.append(cand["nome"])
        print(f"  📉 RESULTADO: ❌ REPROVADO")

# Estatísticas
print("\n" + "=" * 60)
print("📊 RESUMO FINAL:")
print(f"Total candidatos: {len(candidatos)}")
print(f"Aprovados: {len(aprovados)} ({len(aprovados)/len(candidatos)*100:.1f}%)")
print(f"Reprovados: {len(reprovados)} ({len(reprovados)/len(candidatos)*100:.1f}%)")

if aprovados:
    print(f"\n🎯 Candidatos aprovados para entrevista:")
    for i, nome in enumerate(aprovados, 1):
        print(f"  {i}. {nome}")
```

## 🔷 4) **QA: Análise de logs de erro**

### ✅ **Processar logs e identificar padrões**:

```python
# Simulação de logs de erro
logs = [
    "2024-01-15 10:30:15 ERROR Login failed for user admin",
    "2024-01-15 10:31:22 WARN Database connection slow",
    "2024-01-15 10:32:45 ERROR Payment processing failed",
    "2024-01-15 10:33:10 INFO User profile updated",
    "2024-01-15 10:34:55 ERROR Login failed for user test",
    "2024-01-15 10:35:30 ERROR Database timeout",
    "2024-01-15 10:36:05 WARN Cache miss",
    "2024-01-15 10:37:20 ERROR Payment processing failed",
]

print("🔍 ANÁLISE DE LOGS DE ERRO")
print("=" * 60)

# Contadores
contadores = {"ERROR": 0, "WARN": 0, "INFO": 0}
erros_por_tipo = {}
usuarios_com_falha = set()
horas_pico = {}

print("\n📋 Processando logs...")

for log in logs:
    # Extrair informações
    partes = log.split()
    timestamp = partes[0] + " " + partes[1]
    nivel = partes[2]
    mensagem = " ".join(partes[3:])
    
    # Contar por nível
    contadores[nivel] += 1
    
    # Contar erros por tipo
    if nivel == "ERROR":
        # Extrair tipo de erro (primeira palavra após "ERROR")
        tipo_erro = partes[3]
        erros_por_tipo[tipo_erro] = erros_por_tipo.get(tipo_erro, 0) + 1
        
        # Extrair usuário se for falha de login
        if "Login failed for user" in log:
            usuario = partes[6]
            usuarios_com_falha.add(usuario)
    
    # Analisar hora pico (apenas hora)
    hora = partes[1].split(":")[0]
    horas_pico[hora] = horas_pico.get(hora, 0) + 1

# Relatório
print("\n📊 ESTATÍSTICAS GERAIS:")
for nivel, quantidade in contadores.items():
    porcentagem = quantidade / len(logs) * 100
    print(f"  {nivel}: {quantidade} ({porcentagem:.1f}%)")

print("\n🔴 ERROS POR TIPO:")
for tipo, quantidade in sorted(erros_por_tipo.items(), key=lambda x: x[1], reverse=True):
    print(f"  {tipo}: {quantidade}")

if usuarios_com_falha:
    print(f"\n👤 USUÁRIOS COM FALHA DE LOGIN: {', '.join(usuarios_com_falha)}")

# Hora pico
hora_mais_erros = max(horas_pico.items(), key=lambda x: x[1])
print(f"\n⏰ HORA PICO DE ERROS: {hora_mais_erros[0]}h ({hora_mais_erros[1]} logs)")
```
