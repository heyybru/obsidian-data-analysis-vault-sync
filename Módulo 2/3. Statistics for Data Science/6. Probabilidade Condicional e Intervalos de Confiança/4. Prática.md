### 1) Probabilidade condicional com dados simulados

```python
import pandas as pd
import numpy as np

rng = np.random.default_rng(42)

# Dataset simulado: candidatos
n = 500
df = pd.DataFrame({
    "candidate_id": np.arange(1, n+1),
    "canal": rng.choice(["LinkedIn", "Indicação", "Site", "Outro"], size=n, p=[0.5, 0.2, 0.2, 0.1]),
    "chegou_final": rng.choice([0, 1], size=n, p=[0.7, 0.3]),
    "aceitou_oferta": rng.choice([0, 1], size=n, p=[0.85, 0.15])
})

# Ajustar: só pode aceitar se chegou na final
df.loc[df["chegou_final"] == 0, "aceitou_oferta"] = 0

# Tabela cruzada: canal vs chegou_final
tabela = pd.crosstab(df["canal"], df["chegou_final"], margins=True)
print("Tabela cruzada (Canal vs Chegou na Final):\n", tabela)

# Probabilidade condicional: P(chegou_final | Indicação)
p_indicacao = len(df[df["canal"] == "Indicação"]) / len(df)
p_indicacao_e_final = len(df[(df["canal"] == "Indicação") & (df["chegou_final"] == 1)]) / len(df)
p_final_dado_indicacao = p_indicacao_e_final / p_indicacao

print(f"\nP(Chegou na Final | Indicação) = {p_final_dado_indicacao:.2%}")

# Comparar com probabilidade geral
p_geral = df["chegou_final"].mean()
print(f"P(Chegou na Final) geral = {p_geral:.2%}")
```

### 2) Teorema de Bayes aplicado a QA

```python
# Cenário: teste automatizado detecta bugs
# Dados conhecidos:
p_bug = 0.02  # 2% do código tem bugs
p_positivo_dado_bug = 0.95  # sensibilidade (true positive rate)
p_positivo_dado_sem_bug = 0.05  # falso positivo

# Calcular P(positivo)
p_sem_bug = 1 - p_bug
p_positivo = (p_positivo_dado_bug * p_bug) + (p_positivo_dado_sem_bug * p_sem_bug)

# Teorema de Bayes: P(bug | positivo)
p_bug_dado_positivo = (p_positivo_dado_bug * p_bug) / p_positivo

print(f"P(Bug | Teste Positivo) = {p_bug_dado_positivo:.2%}")
print(f"P(Positivo) total = {p_positivo:.2%}")
```

**Interpretação:** mesmo com teste positivo, só há ~28% de chance de bug real (falsos positivos altos + bugs raros).

### 3) Intervalo de confiança para média

```python
from scipy import stats

# Dados: tempo para primeiro contato (horas)
tempos = df["tempo_contato"] = rng.gamma(shape=2.2, scale=6.0, size=n).round(1)

# Estatísticas
media = tempos.mean()
desvio = tempos.std()
n = len(tempos)

# IC 95% para média
ic = stats.t.interval(
    confidence=0.95,
    df=n-1,
    loc=media,
    scale=desvio / np.sqrt(n)
)

print(f"Média amostral: {media:.2f} horas")
print(f"IC 95%: [{ic[0]:.2f}, {ic[1]:.2f}] horas")

# Margem de erro
margem_erro = (ic[1] - ic[0]) / 2
print(f"Margem de erro: ±{margem_erro:.2f} horas")
```

### 4) Intervalo de confiança para proporção

```python
# Proporção de candidatos que aceitaram oferta
sucessos = df["aceitou_oferta"].sum()
n_total = len(df)
proporcao = sucessos / n_total

# IC 95% para proporção (Wilson)
ic_prop = stats.proportion.proportion_confint(
    count=sucessos,
    nobs=n_total,
    method="wilson"
)

print(f"Proporção de aceitação: {proporcao:.2%}")
print(f"IC 95%: [{ic_prop[0]:.2%}, {ic_prop[1]:.2%}]")
```

### 5) Comparando dois grupos com IC

```python
# Comparar tempo de contato por canal
grupo_linkedin = df[df["canal"] == "LinkedIn"]["tempo_contato"]
grupo_indicacao = df[df["canal"] == "Indicação"]["tempo_contato"]

# IC para diferença de médias
media_diff = grupo_indicacao.mean() - grupo_linkedin.mean()

# Erro padrão da diferença
se_diff = np.sqrt(
    grupo_indicacao.var() / len(grupo_indicacao) +
    grupo_linkedin.var() / len(grupo_linkedin)
)

# IC 95%
z = 1.96
ic_diff = (media_diff - z * se_diff, media_diff + z * se_diff)

print(f"Diferença de médias (Indicação - LinkedIn): {media_diff:.2f} horas")
print(f"IC 95% para diferença: [{ic_diff[0]:.2f}, {ic_diff[1]:.2f}]")

# Se IC inclui 0, não há diferença significativa
if ic_diff[0] * ic_diff[1] > 0:
    print("Diferença estatisticamente significativa (IC não inclui 0)")
else:
    print("Sem diferença estatisticamente significativa (IC inclui 0)")
```

### Resumindo

- **Probabilidade condicional**: `crosstab` + divisão de frequências.
- **Bayes**: aplicação direta da fórmula com probabilidades conhecidas.
- **IC para média**: `scipy.stats.t.interval`.
- **IC para proporção**: `stats.proportion.proportion_confint`.
- **Comparação de grupos**: IC para diferença de médias.
